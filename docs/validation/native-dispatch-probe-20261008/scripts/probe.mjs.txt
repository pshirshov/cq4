let n = 0;
const sse = events => events.map(e => `event: ${e.type}\ndata: ${JSON.stringify(e)}\n\n`).join("");
function completed(k, model, items) {
  const base = { id: "resp_" + k, object: "response", model, status: "in_progress", output: [] };
  const usage = { input_tokens: 100 + k, input_tokens_details: { cached_tokens: 0 }, output_tokens: 10 + k, output_tokens_details: { reasoning_tokens: 0 }, total_tokens: 110 + 2 * k };
  const ev = [{ type: "response.created", response: base }];
  items.forEach((item, i) => {
    if (item.type === "message") {
      const text = item.content[0].text;
      ev.push({ type: "response.output_item.added", output_index: i, item: { ...item, content: [], status: "in_progress" } },
        { type: "response.content_part.added", output_index: i, item_id: item.id, content_index: 0, part: { type: "output_text", text: "", annotations: [] } },
        { type: "response.output_text.delta", output_index: i, item_id: item.id, content_index: 0, delta: text },
        { type: "response.output_text.done", output_index: i, item_id: item.id, content_index: 0, text },
        { type: "response.output_item.done", output_index: i, item });
    } else {
      ev.push({ type: "response.output_item.added", output_index: i, item: { ...item, arguments: "", status: "in_progress" } },
        { type: "response.function_call_arguments.delta", output_index: i, item_id: item.id, delta: item.arguments },
        { type: "response.function_call_arguments.done", output_index: i, item_id: item.id, arguments: item.arguments },
        { type: "response.output_item.done", output_index: i, item });
    }
  });
  ev.push({ type: "response.completed", response: { ...base, status: "completed", output: items, usage } });
  return sse(ev);
}
const text = (k, t) => ({ id: "msg_" + k, type: "message", role: "assistant", status: "completed", content: [{ type: "output_text", text: t, annotations: [] }] });
const call = (k, name, args) => ({ id: "fc_" + k, type: "function_call", status: "completed", call_id: "call_" + k, name, arguments: JSON.stringify(args) });

function lastUserText(input) {
  for (let i = input.length - 1; i >= 0; i--) {
    const it = input[i];
    if (it.role === "user") { const t = JSON.stringify(it.content); if (!t.includes("[subagents]")) return { i, text: t }; }
  }
  return { i: -1, text: "" };
}
function outputsOf(input, name) {
  const callIds = new Map(input.filter(x => x.type === "function_call").map(x => [x.call_id, x.name]));
  return input.filter(x => x.type === "function_call_output" && callIds.get(x.call_id) === name).map(x => String(x.output));
}
async function script(req, k) {
  const tools = (req.tools || []).map(t => t.name);
  const input = req.input || [];
  const { i: ui, text: user } = lastUserText(input);
  const after = input.slice(ui + 1).filter(x => x.type === "function_call").map(x => x.name);
  const model = req.model;
  const has = n => after.filter(x => x === n).length;
  const idOf = s => JSON.parse(s.replace(/^[^{]*/, "")).agentId;
  if (tools.includes("spawn_agent")) {                // governing session
    const spawns = outputsOf(input.slice(ui + 1), "spawn_agent").map(idOf);
    if (user.includes("SCENARIO-A")) {
      if (has("spawn_agent") === 0) return [call(k, "spawn_agent", { task_name: "reader1", profile: "reader", message: "READER-TASK: read data.txt and report its first line." })];
      if (has("spawn_agent") === 1) return [call(k, "spawn_agent", { task_name: "writer1", profile: "writer", message: "WRITER-TASK: create out.txt containing 'written by writer1'." })];
      if (!has("wait_agent")) return [call(k, "wait_agent", { condition: "all_settled", timeout_ms: 60000 })];
      if (!has("list_agents")) return [call(k, "list_agents", {})];
      if (has("read_agent") < 2) return [call(k, "read_agent", { target: spawns[has("read_agent")], view: "result" })];
      return [text(k, "A-DONE")];
    }
    if (user.includes("SCENARIO-B")) {
      if (!has("spawn_agent")) return [call(k, "spawn_agent", { task_name: "slow", profile: "reader", message: "READER-TASK delayed: read data.txt." })];
      return [text(k, "B-SPAWNED: not waiting")];
    }
    if (user.includes("SCENARIO-C")) {
      if (!has("spawn_agent")) return [call(k, "spawn_agent", { task_name: "gated", profile: "writer", message: "WRITER-TASK: create out.txt containing 'gated'.",
        gate: { model: { provider: "openai", id: model }, thinkingLevel: "low", prompt: "Approve when out.txt exists.", maxRounds: 2,
          checks: [{ id: "exists", command: "test -f out.txt && echo present", timeoutMs: 20000 }], promisedOutputs: ["out.txt"] } })];
      if (!has("wait_agent")) return [call(k, "wait_agent", { condition: "all_settled", timeout_ms: 90000 })];
      if (!has("list_agents")) return [call(k, "list_agents", {})];
      if (!has("read_agent")) return [call(k, "read_agent", { target: spawns[0], view: "events", limit: 100 })];
      return [text(k, "C-DONE")];
    }
    if (user.includes("SCENARIO-D")) {
      if (!has("spawn_agent")) return [call(k, "spawn_agent", { task_name: "boxed", profile: "writer", isolation: "sandbox", repo_id: "proj", message: "WRITER-TASK: create out.txt containing 'boxed'." })];
      if (!has("wait_agent")) return [call(k, "wait_agent", { condition: "all_settled", timeout_ms: 60000 })];
      if (!has("read_agent")) return [call(k, "read_agent", { target: spawns[0], view: "result" })];
      return [text(k, "D-DONE")];
    }
    if (user.includes("SCENARIO-E")) {
      if (!has("spawn_agent")) return [call(k, "spawn_agent", { task_name: "victim", profile: "reader", message: "READER-TASK delayed: read data.txt." })];
      if (!has("interrupt_agent")) return [call(k, "interrupt_agent", { target: spawns[0] })];
      if (!has("wait_agent")) return [call(k, "wait_agent", { condition: "all_settled", timeout_ms: 30000 })];
      if (!has("read_agent")) return [call(k, "read_agent", { target: spawns[0], view: "status" })];
      return [text(k, "E-DONE")];
    }
    if (user.includes("SCENARIO-F")) {
      if (!has("probe_bridge")) return [call(k, "probe_bridge", { profile: "writer", message: "WRITER-TASK: create out.txt containing 'bridged'." })];
      return [text(k, "F-DONE")];
    }
    return [text(k, "ack")];
  }
  if (tools.includes("submit_gate_decision")) {       // managed gate reviewer
    const m = /candidateId \\\\?"([^"\\\\]+)/.exec(user);
    if (!after.length) return [call(k, "grep", { pattern: "gated", path: "out.txt", literal: true })];
    return [call(k, "submit_gate_decision", { schemaVersion: 1, candidateId: m ? m[1] : "?", decision: "approve", advisories: [] })];
  }
  if (tools.includes("write")) {                      // writer child
    if (after.length === 0) return [call(k, "write", { path: "out.txt", content: user.includes("gated") ? "gated\n" : "written by writer1\n" })];
    if (after.length === 1) return [call(k, "bash", { command: "git status --short && pwd && git rev-parse HEAD" })];
    return [text(k, "WRITER-RESULT: wrote out.txt")];
  }
  if (after.length === 0) { if (user.includes("delayed")) await sleep(5000); return [call(k, "read", { path: "data.txt" })]; }   // reader child
  return [text(k, "READER-RESULT: data.txt first line is 'alpha'")];
}
const provider = createServer((rq, rs) => {
  let raw = "";
  rq.on("data", c => raw += c);
  rq.on("end", async () => {
    let req = {};
    try { req = JSON.parse(raw); } catch {}
    const k = ++n;
    let items;
    try { items = await script(req, k); } catch (e) { items = [text(k, "SCRIPT-ERROR " + e)]; }
    const tools = (req.tools || []).map(t => t.name);
    appendFileSync(log, JSON.stringify({ n: k, at: new Date().toISOString(), path: rq.url, model: req.model, tools, reply: items.map(i => i.type === "function_call" ? `call ${i.name} ${i.arguments.slice(0, 200)}` : "text " + i.content[0].text),
      inputTail: (req.input || []).slice(-2).map(i => (i.role || i.type) + ": " + JSON.stringify(i.content ?? i.output ?? i.arguments).slice(0, 600)),
      noticeInInput: JSON.stringify(req.input || []).includes("[subagents]") }) + "\n");
    const body = completed(k, req.model || "stub", items);
    rs.writeHead(200, { "content-type": "text/event-stream", "content-length": Buffer.byteLength(body) });
    rs.end(body);
  });
});
await new Promise(r => provider.listen(0, "127.0.0.1", r));

async function until(what, pred, from, ms = 120000) {
  const t0 = Date.now();
  for (;;) { const i = records.findIndex((r, ix) => ix >= from && pred(r)); if (i >= 0) return i; if (Date.now() - t0 > ms) throw new Error("timeout: " + what + "\n" + stderr); await sleep(50); }
}
async function cmd(value) {
  const id = randomUUID(); const from = records.length;
  child.stdin.write(JSON.stringify({ id, ...value }) + "\n");
  const i = await until(value.type, r => r.type === "response" && r.id === id, from);
  return records[i];
}
async function turn(message) {
  const from = records.length;
  await cmd({ type: "prompt", message });
  const i = await until("agent_settled for " + message, r => r.type === "agent_settled", from, 180000);
  return { from, end: i };
}
